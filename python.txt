# 1. Número positivo ou negativo
num1 = int(input("1. Digite um número: "))

    print("Negativo")

# 12. Soma e comparação de dois números
n1_12 = float(input("\n12. Número 1: "))

print(f"Soma: {n1_12 + n2_12}")
if n1_12 > n2_12:

# 17. Faixa etária
idade = int(input("\n17. Digite sua idade: "))
if idade < 18:

elif 18 <= idade <= 59:

else:

# 18. Classificação completa
num18 = int(input("\n18. Digite um número: "))
if num18 == 0:
    print("Neutro")
else:
    p_i = "par" if num18 % 2 == 0 else "impar"
    p_n = "positivo" if num18 > 0 else "negativo"
    print(f"{p_i} {p_n}")

# 19. Iguais ou Diferentes
n1_19 = float(input("\n19. Valor 1: "))

if n1_19 == n2_19:
    print("São iguais")
else:
    print(f"Diferentes. Diferença: {abs(n1_19 - n2_19)}")
